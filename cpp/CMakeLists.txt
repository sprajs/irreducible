# Reference arithmetic only; never a production dependency or default test.
option(IRRED_REFERENCE_GMP_MPFR "Enable optional directed-interval BAO reference test" OFF)
if(IRRED_REFERENCE_GMP_MPFR)
  find_path(IRRED_GMP_INCLUDE_DIR gmp.h)
  find_path(IRRED_GMPXX_INCLUDE_DIR gmpxx.h)
  find_path(IRRED_MPFR_INCLUDE_DIR mpfr.h)
  find_library(IRRED_GMP_LIBRARY NAMES gmp)
  find_library(IRRED_GMPXX_LIBRARY NAMES gmpxx)
  find_library(IRRED_MPFR_LIBRARY NAMES mpfr)
  if(NOT IRRED_GMP_INCLUDE_DIR OR NOT IRRED_GMPXX_INCLUDE_DIR OR NOT IRRED_MPFR_INCLUDE_DIR OR NOT IRRED_GMP_LIBRARY OR NOT IRRED_GMPXX_LIBRARY OR NOT IRRED_MPFR_LIBRARY)
    message(FATAL_ERROR "Directed interval reference unavailable: installed GMP/GMPXX/MPFR headers and dynamic libraries are required")
  endif()
  foreach(reference_library IN ITEMS IRRED_GMP_LIBRARY IRRED_GMPXX_LIBRARY IRRED_MPFR_LIBRARY)
    get_filename_component(reference_real_path "${${reference_library}}" REALPATH)
    if(NOT reference_real_path MATCHES "\\.so(\\.|$)")
      message(FATAL_ERROR "Directed interval reference requires dynamic system libraries: ${reference_real_path}")
    endif()
  endforeach()
  if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_bao_piecewise_interval.cpp")
    message(FATAL_ERROR "Directed interval reference unavailable: reference source has not been authored")
  endif()
  add_executable(test_bao_piecewise_interval EXCLUDE_FROM_ALL tests/test_bao_piecewise_interval.cpp)
  target_include_directories(test_bao_piecewise_interval PRIVATE ${IRRED_GMP_INCLUDE_DIR} ${IRRED_GMPXX_INCLUDE_DIR} ${IRRED_MPFR_INCLUDE_DIR})
  target_compile_options(test_bao_piecewise_interval PRIVATE -Wall -Wextra -Wpedantic -fno-fast-math -ffp-contract=off)
  target_link_libraries(test_bao_piecewise_interval PRIVATE irred_core ${IRRED_MPFR_LIBRARY} ${IRRED_GMPXX_LIBRARY} ${IRRED_GMP_LIBRARY})
  add_test(NAME bao_piecewise_interval_controls COMMAND test_bao_piecewise_interval)
  message(STATUS "Directed interval reference enabled as explicit test_bao_piecewise_interval target; no default external-data test")
else()
  message(STATUS "Directed interval reference not enabled; normal production build has no GMP/MPFR reference dependency")
endif()
