cmake_minimum_required(VERSION 3.24)
project(cosmology_core VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
add_library(cosmology_core STATIC src/abi.cpp src/quantities.cpp src/numerics.cpp)
target_include_directories(cosmology_core PUBLIC include)
target_compile_options(cosmology_core PRIVATE -Wall -Wextra -Wpedantic -fno-fast-math -ffp-contract=off)
add_executable(test_abi tests/test_abi.cpp)
target_link_libraries(test_abi PRIVATE cosmology_core)
enable_testing()
add_test(NAME abi_contract COMMAND test_abi)
install(TARGETS cosmology_core ARCHIVE DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
add_executable(test_abi_hostile tests/test_abi_hostile.cpp)
target_link_libraries(test_abi_hostile PRIVATE cosmology_core)
add_test(NAME abi_hostile_contract COMMAND test_abi_hostile)

add_executable(test_quantities tests/test_quantities.cpp)
target_link_libraries(test_quantities PRIVATE cosmology_core)
add_test(NAME quantities_contract COMMAND test_quantities)

add_executable(test_quantities_hostile tests/test_quantities_hostile.cpp)
target_link_libraries(test_quantities_hostile PRIVATE cosmology_core)
add_test(NAME quantities_hostile_contract COMMAND test_quantities_hostile)
add_executable(test_quantity_abi_hostile tests/test_quantity_abi_hostile.cpp)
target_link_libraries(test_quantity_abi_hostile PRIVATE cosmology_core)
add_test(NAME quantity_abi_hostile_contract COMMAND test_quantity_abi_hostile)

add_executable(test_numerics tests/test_numerics.cpp)
target_link_libraries(test_numerics PRIVATE cosmology_core)
add_test(NAME numerics_contract COMMAND test_numerics)

add_executable(test_numerics_hostile tests/test_numerics_hostile.cpp)
target_link_libraries(test_numerics_hostile PRIVATE cosmology_core)
add_test(NAME numerics_hostile_contract COMMAND test_numerics_hostile)
